' ==============================================================================
' RULE: B1_Align_Selected_Face_To_XZ_Ground (V16.0 KeyShot Super Bright Studio & Auto-Clean Engine)
' Tác giả: Phan Trọng Tấn (@tanphan1105) - WaterTreatment3D
' Chức năng:
'   1. HỆ THỐNG ÁNH SÁNG SIÊU SÁNG (Ambience 65%, Key Light 220%, Rim Light 250%)
'   2. TỰ ĐỘNG QUÉT & ẨN SẠCH 100% SKETCHES 2D/3D, SURFACE BODIES, WORKSURFACES, WORKPLANES
'   3. Khắc phục triệt để lỗi nhân đôi ma trận FaceProxy trong Sub-Assembly
'   4. Hỗ trợ chọn cả Mặt phẳng Solid (Face) lẫn Mặt phẳng làm việc (WorkPlane)
'   5. Thuật toán kiểm định 2 bước: Xoay phẳng mặt đáy -> Tự động đưa thân máy lên +Y
'   6. TIẾP ĐẤT Y=0 VÀ CĂN TÂM ĐỐI XỨNG (X=0, Z=0) CHÍNH XÁC 100%
'   7. Tự động tắt sạch iMate Glyphs, Joint Glyphs, Work Features
'   8. Thiết lập ViewCube và Camera Perspective 35mm chuẩn studio KeyShot
'   9. Hỗ trợ hoàn hảo 100% CẢ FILE LẮP RÁP (.IAM) và CHI TIẾT (.IPT)
' ==============================================================================

Imports System
Imports System.Collections
Imports System.Collections.Generic
Imports Inventor

Sub Main()
    Dim oDoc As Document = ThisApplication.ActiveDocument
    If oDoc Is Nothing Then
        MessageBox.Show("Không có file nào đang mở!", "WT3D", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        Return
    End If

    Dim tg As TransientGeometry = ThisApplication.TransientGeometry

    ' --------------------------------------------------------------------------
    ' 1. MENU ĐIỀU HƯỚNG BẰNG NATIVE INPUTLISTBOX
    ' --------------------------------------------------------------------------
    Dim optList As New ArrayList
    optList.Add("1. 🎯 [TIẾP ĐẤT & BẬT SUPER BRIGHT STUDIO]: Ép tiếp đất Y=0, thân +Y, căn tâm & tự động bật Studio Siêu Sáng + Dọn sạch Sketch/Surface")
    optList.Add("2. 🤖 [AUTO SMART]: Tự động nhận diện trục dài nhất đưa về +Y")
    optList.Add("3. 🔄 [XOAY X +90°]: Xoay quanh trục X (+90°)")
    optList.Add("4. 🔄 [XOAY X -90°]: Xoay quanh trục X (-90°)")
    optList.Add("5. 🔄 [XOAY Z +90°]: Xoay quanh trục Z (+90°)")
    optList.Add("6. 🔄 [XOAY Z -90°]: Xoay quanh trục Z (-90°)")
    optList.Add("7. 🔃 [ĐẢO ĐẦU 180°]: Lật ngược 180° nếu đang bị chúc đầu")
    optList.Add("8. 🧭 [CHỈ CĂN TÂM & ĐÁY]: Đặt đáy về Y=0 và tâm (0,0,0) không xoay")
    optList.Add("9. 🌟 [KEYSHOT SUPER BRIGHT STUDIO]: 4 Đèn Siêu Sáng (Key 220% + Rim 250%) + Sàn Gương + Ẩn Sketch/Surface (xem ngay)")
    optList.Add("10. 💎 [KEYSHOT METALLIC SHINE (Inox)]: Tương phản cực mạnh, vệt highlight rực sáng cho kim loại (xem ngay)")
    optList.Add("11. 💡 [CLEAN CAD STUDIO]: Default Lights trắng sáng kỹ thuật (xem ngay)")

    Dim sel As String = InputListBox("Chọn chế độ căn chỉnh trục đứng (Y-Up) & tiếp đất sàn Y=0:", optList, optList(0), "🧭 WT3D B1 ALIGNER V16.0", "Danh sách thao tác:")
    If String.IsNullOrEmpty(sel) Then Return

    ' ==============================================================================
    ' TRƯỜNG HỢP A: FILE LẮP RÁP (.IAM - ASSEMBLY DOCUMENT)
    ' ==============================================================================
    If oDoc.DocumentType = DocumentTypeEnum.kAssemblyDocumentObject Then
        Dim oAsmDoc As AssemblyDocument = CType(oDoc, AssemblyDocument)
        Dim oDef As AssemblyComponentDefinition = oAsmDoc.ComponentDefinition

        If oDef.Occurrences.Count = 0 Then
            MessageBox.Show("Cụm lắp ráp rỗng!", "WT3D", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            Return
        End If

        ' Tạm ngắt constraints & joints
        For Each c As AssemblyConstraint In oDef.Constraints
            Try : c.Suppressed = True : Catch : End Try
        Next
        For Each j As AssemblyJoint In oDef.Joints
            Try : j.Suppressed = True : Catch : End Try
        Next

        ' Cache grounded
        For Each occ As ComponentOccurrence In oDef.Occurrences
            If Not occ.Suppressed Then
                Try : occ.Grounded = False : Catch : End Try
            End If
        Next

        Dim rotMatrix As Matrix = tg.CreateMatrix()
        Dim doRotate As Boolean = True
        Dim planeGeom As Plane = Nothing
        Dim rawNormal As Vector = Nothing
        Dim ptOnPlane As Point = Nothing

        ' --- A1. CHỌN MẶT ĐÁY (FACE HOẶC WORKPLANE) ---
        If sel.StartsWith("1.") Then
            Dim selectedObj As Object = Nothing

            ' Kiểm tra đối tượng đã chọn sẵn trong SelectSet
            If oDoc.SelectSet.Count > 0 Then
                For Each obj As Object In oDoc.SelectSet
                    If TypeOf obj Is Face OrElse TypeOf obj Is WorkPlane OrElse TypeOf obj Is WorkPlaneProxy Then
                        selectedObj = obj
                        Exit For
                    End If
                Next
            End If

            ' Nếu chưa chọn sẵn -> mở Pick
            If selectedObj Is Nothing Then
                Try
                    selectedObj = ThisApplication.CommandManager.Pick(SelectionFilterEnum.kAllEntitiesFilter, "👉 CLICK CHỌN MẶT ĐÁY (SOLID FACE) HOẶC WORKPLANE (TRÊN 3D HOẶC MODEL TREE):")
                Catch
                    Return
                End Try
            End If

            If selectedObj Is Nothing Then Return

            ' Trích xuất thông tin hình học chuẩn xác từ Face / FaceProxy / WorkPlane
            If TypeOf selectedObj Is Face Then
                Dim f As Face = CType(selectedObj, Face)
                If f.SurfaceType <> SurfaceTypeEnum.kPlaneSurface Then
                    MessageBox.Show("Mặt được chọn không phải là mặt phẳng!", "WT3D", MessageBoxButtons.OK, MessageBoxIcon.Error)
                    Return
                End If

                ' LƯU Ý KỸ THUẬT: Trong Inventor Assembly, f.Geometry của FaceProxy đã ở hệ tọa độ Top Assembly!
                planeGeom = CType(f.Geometry, Plane)
                rawNormal = planeGeom.Normal.AsVector()
                ptOnPlane = f.PointOnFace

            ElseIf TypeOf selectedObj Is WorkPlane Then
                Dim wp As WorkPlane = CType(selectedObj, WorkPlane)
                planeGeom = wp.Plane
                rawNormal = planeGeom.Normal.AsVector()
                ptOnPlane = planeGeom.RootPoint

            ElseIf TypeOf selectedObj Is WorkPlaneProxy Then
                Dim wpProxy As WorkPlaneProxy = CType(selectedObj, WorkPlaneProxy)
                planeGeom = wpProxy.Plane
                rawNormal = planeGeom.Normal.AsVector()
                ptOnPlane = planeGeom.RootPoint
            Else
                MessageBox.Show("Đối tượng được chọn không phải là Mặt phẳng hoặc WorkPlane!", "WT3D", MessageBoxButtons.OK, MessageBoxIcon.Warning)
                Return
            End If

            ' Chuẩn hóa vector pháp tuyến
            Dim lenN As Double = Math.Sqrt(rawNormal.X * rawNormal.X + rawNormal.Y * rawNormal.Y + rawNormal.Z * rawNormal.Z)
            If lenN < 0.000001 Then lenN = 1.0
            Dim nx As Double = rawNormal.X / lenN
            Dim ny As Double = rawNormal.Y / lenN
            Dim nz As Double = rawNormal.Z / lenN

            ' Đưa pháp tuyến mặt phẳng về trục Y (nằm ngang song song mặt đất XZ)
            Dim targetVec As Vector = tg.CreateVector(0, -1, 0)
            Dim dot As Double = nx * targetVec.X + ny * targetVec.Y + nz * targetVec.Z

            If Math.Abs(dot - 1.0) > 0.0001 Then
                Dim rotAxis As Vector = Nothing
                Dim rotAngle As Double = 0.0

                If Math.Abs(dot - (-1.0)) < 0.0001 Then
                    rotAxis = tg.CreateVector(1, 0, 0)
                    rotAngle = Math.PI
                Else
                    Dim crossX As Double = ny * targetVec.Z - nz * targetVec.Y
                    Dim crossY As Double = nz * targetVec.X - nx * targetVec.Z
                    Dim crossZ As Double = nx * targetVec.Y - ny * targetVec.X
                    Dim crossLen As Double = Math.Sqrt(crossX * crossX + crossY * crossY + crossZ * crossZ)
                    If crossLen > 0.0001 Then
                        rotAxis = tg.CreateVector(crossX / crossLen, crossY / crossLen, crossZ / crossLen)
                    Else
                        rotAxis = tg.CreateVector(1, 0, 0)
                    End If
                    Dim clampedDot As Double = Math.Max(-1.0, Math.Min(1.0, dot))
                    rotAngle = Math.Acos(clampedDot)
                End If

                rotMatrix.SetToRotation(rotAngle, rotAxis, tg.CreatePoint(0, 0, 0))
            Else
                doRotate = False
            End If

        ' --- A2. CÁC LỰA CHỌN XOAY KHÁC ---
        ElseIf sel.StartsWith("2.") Then
            Dim minPt As Point = oDef.RangeBox.MinPoint
            Dim maxPt As Point = oDef.RangeBox.MaxPoint
            Dim sx As Double = maxPt.X - minPt.X
            Dim sy As Double = maxPt.Y - minPt.Y
            Dim sz As Double = maxPt.Z - minPt.Z
            If sx >= sy AndAlso sx >= sz Then
                rotMatrix.SetToRotation(Math.PI / 2, tg.CreateVector(0, 0, 1), tg.CreatePoint(0, 0, 0))
            ElseIf sz >= sx AndAlso sz >= sy Then
                rotMatrix.SetToRotation(-Math.PI / 2, tg.CreateVector(1, 0, 0), tg.CreatePoint(0, 0, 0))
            Else
                doRotate = False
            End If
        ElseIf sel.StartsWith("3.") Then
            rotMatrix.SetToRotation(Math.PI / 2, tg.CreateVector(1, 0, 0), tg.CreatePoint(0, 0, 0))
        ElseIf sel.StartsWith("4.") Then
            rotMatrix.SetToRotation(-Math.PI / 2, tg.CreateVector(1, 0, 0), tg.CreatePoint(0, 0, 0))
        ElseIf sel.StartsWith("5.") Then
            rotMatrix.SetToRotation(Math.PI / 2, tg.CreateVector(0, 0, 1), tg.CreatePoint(0, 0, 0))
        ElseIf sel.StartsWith("6.") Then
            rotMatrix.SetToRotation(-Math.PI / 2, tg.CreateVector(0, 0, 1), tg.CreatePoint(0, 0, 0))
        ElseIf sel.StartsWith("7.") Then
            rotMatrix.SetToRotation(Math.PI, tg.CreateVector(1, 0, 0), tg.CreatePoint(0, 0, 0))
        ElseIf sel.StartsWith("8.") OrElse sel.StartsWith("9.") OrElse sel.StartsWith("10.") OrElse sel.StartsWith("11.") Then
            doRotate = False
        End If

        ' Áp dụng xoay ban đầu cho toàn bộ Occurrences
        If doRotate Then
            For Each occ As ComponentOccurrence In oDef.Occurrences
                If Not occ.Suppressed Then
                    Try
                        Dim curMat As Matrix = occ.Transformation
                        curMat.PreMultiplyBy(rotMatrix)
                        occ.Transformation = curMat
                    Catch : End Try
                End If
            Next
            oAsmDoc.Update()
        End If

        ' KIỂM TRA CHỐNG CHÚC ĐẦU NGƯỢC CHIỀU (+Y UP GUARD CHO ASSEMBLY):
        If sel.StartsWith("1.") AndAlso ptOnPlane IsNot Nothing Then
            Dim curMinY As Double = 1.0E+15, curMaxY As Double = -1.0E+15
            For Each occ As ComponentOccurrence In oDef.Occurrences
                If Not occ.Suppressed Then
                    Try
                        Dim rBox As Box = occ.RangeBox
                        If rBox IsNot Nothing Then
                            If rBox.MinPoint.Y < curMinY Then curMinY = rBox.MinPoint.Y
                            If rBox.MaxPoint.Y > curMaxY Then curMaxY = rBox.MaxPoint.Y
                        End If
                    Catch : End Try
                End If
            Next

            If doRotate Then
                ptOnPlane.TransformBy(rotMatrix)
            End If

            Dim faceY As Double = ptOnPlane.Y
            Dim centerY As Double = (curMinY + curMaxY) / 2.0

            ' Nếu trọng tâm thấp hơn mặt đáy -> máy đang bị chúc ngược đầu -> lật 180° quanh trục X!
            If centerY < (faceY - 0.01) Then
                Dim flipMat As Matrix = tg.CreateMatrix()
                flipMat.SetToRotation(Math.PI, tg.CreateVector(1, 0, 0), tg.CreatePoint(0, 0, 0))
                For Each occ As ComponentOccurrence In oDef.Occurrences
                    If Not occ.Suppressed Then
                        Try
                            Dim curMat As Matrix = occ.Transformation
                            curMat.PreMultiplyBy(flipMat)
                            occ.Transformation = curMat
                        Catch : End Try
                    End If
                Next
                oAsmDoc.Update()
            End If
        End If

        ' ĐO BOUNDING BOX VÀ ÉP TIẾP ĐẤT Y=0 CHUẨN XÁC 100%
        If Not sel.StartsWith("9.") AndAlso Not sel.StartsWith("10.") AndAlso Not sel.StartsWith("11.") Then
            Dim minX As Double = 1.0E+15, minY As Double = 1.0E+15, minZ As Double = 1.0E+15
            Dim maxX As Double = -1.0E+15, maxY As Double = -1.0E+15, maxZ As Double = -1.0E+15

            For Each occ As ComponentOccurrence In oDef.Occurrences
                If Not occ.Suppressed Then
                    Try
                        Dim rBox As Box = occ.RangeBox
                        If rBox IsNot Nothing Then
                            If rBox.MinPoint.X < minX Then minX = rBox.MinPoint.X
                            If rBox.MinPoint.Y < minY Then minY = rBox.MinPoint.Y
                            If rBox.MinPoint.Z < minZ Then minZ = rBox.MinPoint.Z
                            If rBox.MaxPoint.X > maxX Then maxX = rBox.MaxPoint.X
                            If rBox.MaxPoint.Y > maxY Then maxY = rBox.MaxPoint.Y
                            If rBox.MaxPoint.Z > maxZ Then maxZ = rBox.MaxPoint.Z
                        End If
                    Catch : End Try
                End If
            Next

            Dim cx As Double = (maxX + minX) / 2.0
            Dim cz As Double = (maxZ + minZ) / 2.0
            Dim bottomY As Double = minY

            For Each occ As ComponentOccurrence In oDef.Occurrences
                If Not occ.Suppressed Then
                    Try
                        Dim mat As Matrix = occ.Transformation
                        mat.Cell(1, 4) = mat.Cell(1, 4) - cx
                        mat.Cell(2, 4) = mat.Cell(2, 4) - bottomY
                        mat.Cell(3, 4) = mat.Cell(3, 4) - cz
                        occ.Transformation = mat
                        occ.Grounded = True
                    Catch : End Try
                End If
            Next

            oAsmDoc.Update()
        End If

    ' ==============================================================================
    ' TRƯỜNG HỢP B: FILE CHI TIẾT (.IPT - PART DOCUMENT)
    ' ==============================================================================
    ElseIf oDoc.DocumentType = DocumentTypeEnum.kPartDocumentObject Then
        Dim oPartDoc As PartDocument = CType(oDoc, PartDocument)
        Dim oCompDef As PartComponentDefinition = oPartDoc.ComponentDefinition

        If oCompDef.SurfaceBodies.Count = 0 Then
            MessageBox.Show("Chi tiết không có Solid Body!", "WT3D", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            Return
        End If

        Dim bodies As ObjectCollection = ThisApplication.TransientObjects.CreateObjectCollection()
        For Each sb As SurfaceBody In oCompDef.SurfaceBodies
            bodies.Add(sb)
        Next

        ' Xử lý chọn mặt đáy (Face hoặc WorkPlane) cho Part
        If sel.StartsWith("1.") Then
            Dim selectedObj As Object = Nothing
            If oDoc.SelectSet.Count > 0 Then
                For Each obj As Object In oDoc.SelectSet
                    If TypeOf obj Is Face OrElse TypeOf obj Is WorkPlane Then
                        selectedObj = obj
                        Exit For
                    End If
                Next
            End If

            If selectedObj Is Nothing Then
                Try
                    selectedObj = ThisApplication.CommandManager.Pick(SelectionFilterEnum.kAllEntitiesFilter, "👉 CLICK CHỌN MẶT ĐÁY (SOLID FACE) HOẶC WORKPLANE (TRÊN 3D HOẶC MODEL TREE):")
                Catch
                    Return
                End Try
            End If

            If selectedObj Is Nothing Then Return

            Dim rawNormal As Vector = Nothing
            Dim ptOnPlane As Point = Nothing

            If TypeOf selectedObj Is Face Then
                Dim f As Face = CType(selectedObj, Face)
                If f.SurfaceType <> SurfaceTypeEnum.kPlaneSurface Then
                    MessageBox.Show("Mặt được chọn không phải là mặt phẳng!", "WT3D", MessageBoxButtons.OK, MessageBoxIcon.Error)
                    Return
                End If
                Dim planeGeom As Plane = CType(f.Geometry, Plane)
                rawNormal = planeGeom.Normal.AsVector()
                ptOnPlane = f.PointOnFace

            ElseIf TypeOf selectedObj Is WorkPlane Then
                Dim wp As WorkPlane = CType(selectedObj, WorkPlane)
                Dim planeGeom As Plane = wp.Plane
                rawNormal = planeGeom.Normal.AsVector()
                ptOnPlane = planeGeom.RootPoint
            Else
                MessageBox.Show("Đối tượng được chọn không phải là Mặt phẳng hoặc WorkPlane!", "WT3D", MessageBoxButtons.OK, MessageBoxIcon.Warning)
                Return
            End If

            Dim nx As Double = rawNormal.X, ny As Double = rawNormal.Y, nz As Double = rawNormal.Z
            Dim lenN As Double = Math.Sqrt(nx * nx + ny * ny + nz * nz)
            If lenN < 0.000001 Then lenN = 1.0
            nx /= lenN : ny /= lenN : nz /= lenN

            Dim dot As Double = nx * 0.0 + ny * (-1.0) + nz * 0.0

            If Math.Abs(dot - 1.0) > 0.0001 Then
                Dim rotAxisVec As UnitVector = Nothing
                Dim rotAngleRad As Double = 0.0
                If Math.Abs(dot - (-1.0)) < 0.0001 Then
                    rotAxisVec = tg.CreateUnitVector(1, 0, 0)
                    rotAngleRad = Math.PI
                Else
                    Dim crossX As Double = ny * 0.0 - nz * (-1.0)
                    Dim crossY As Double = nz * 0.0 - nx * 0.0
                    Dim crossZ As Double = nx * (-1.0) - ny * 0.0
                    Dim crossLen As Double = Math.Sqrt(crossX * crossX + crossY * crossY + crossZ * crossZ)
                    If crossLen > 0.0001 Then
                        rotAxisVec = tg.CreateUnitVector(crossX / crossLen, crossY / crossLen, crossZ / crossLen)
                    Else
                        rotAxisVec = tg.CreateUnitVector(1, 0, 0)
                    End If
                    rotAngleRad = Math.Acos(Math.Max(-1.0, Math.Min(1.0, dot)))
                End If

                Dim tempAxis As WorkAxis = oCompDef.WorkAxes.AddFixed(tg.CreatePoint(0, 0, 0), rotAxisVec)
                tempAxis.Visible = False
                Dim moveDef1 As MoveDefinition = oCompDef.Features.MoveFeatures.CreateMoveDefinition(bodies)
                moveDef1.AddRotateAboutAxis(tempAxis, True, (rotAngleRad * 180.0 / Math.PI).ToString() & " deg")
                oCompDef.Features.MoveFeatures.Add(moveDef1)
                Try : tempAxis.Delete() : Catch : End Try
                oPartDoc.Update()
            End If

            ' Kiểm tra chống chúc đầu ngược cho Part
            Dim pMin1 As Point = oCompDef.RangeBox.MinPoint
            Dim pMax1 As Point = oCompDef.RangeBox.MaxPoint
            Dim center1Y As Double = (pMin1.Y + pMax1.Y) / 2.0
            If center1Y < (ptOnPlane.Y - 0.01) Then
                Dim tempAxisX As WorkAxis = oCompDef.WorkAxes.AddFixed(tg.CreatePoint(0, 0, 0), tg.CreateUnitVector(1, 0, 0))
                tempAxisX.Visible = False
                Dim moveDefFlip As MoveDefinition = oCompDef.Features.MoveFeatures.CreateMoveDefinition(bodies)
                moveDefFlip.AddRotateAboutAxis(tempAxisX, True, "180 deg")
                oCompDef.Features.MoveFeatures.Add(moveDefFlip)
                Try : tempAxisX.Delete() : Catch : End Try
                oPartDoc.Update()
            End If
        End If

        ' Tiếp đất Y=0 và căn tâm cho Part
        If Not sel.StartsWith("9.") AndAlso Not sel.StartsWith("10.") AndAlso Not sel.StartsWith("11.") Then
            Dim pMin As Point = oCompDef.RangeBox.MinPoint
            Dim pMax As Point = oCompDef.RangeBox.MaxPoint
            Dim dX As Double = - (pMin.X + pMax.X) / 2.0
            Dim dY As Double = - pMin.Y
            Dim dZ As Double = - (pMin.Z + pMax.Z) / 2.0

            Dim moveDef2 As MoveDefinition = oCompDef.Features.MoveFeatures.CreateMoveDefinition(bodies)
            moveDef2.AddFreeDrag(dX.ToString() & " cm", dY.ToString() & " cm", dZ.ToString() & " cm")
            oCompDef.Features.MoveFeatures.Add(moveDef2)
            oPartDoc.Update()
        End If
    End If

    ' --------------------------------------------------------------------------
    ' 3. THIẾT LẬP HỆ THỐNG 4 NGUỒN ĐÈN KEYSHOT SUPER BRIGHT STUDIO RIG
    ' --------------------------------------------------------------------------
    Try
        Dim targetStyleName As String = "Default Lights"
        If sel.StartsWith("10.") OrElse sel.Contains("METALLIC SHINE") Then
            targetStyleName = "Stark Highlights"
        ElseIf sel.StartsWith("11.") OrElse sel.Contains("CLEAN CAD") Then
            targetStyleName = "Default Lights"
        Else
            targetStyleName = "Two Lights"
        End If

        Try
            oDoc.ActiveLightingStyle = oDoc.LightingStyles.Item(targetStyleName)
        Catch
            Try : oDoc.ActiveLightingStyle = oDoc.LightingStyles.Item("Default Lights") : Catch : End Try
        End Try

        ' TINH CHỈNH ĐỘ SÁNG BỪNG VÀ HỆ THỐNG ĐÈN 4 HƯỚNG CHUẨN KEYSHOT (HIGH-KEY COMMERCIAL)
        Try
            Dim curStyle As LightingStyle = oDoc.ActiveLightingStyle
            If curStyle IsNot Nothing Then
                ' Nâng Ambience lên 0.65 để xóa tan hoàn toàn bóng tối, toàn bộ mô hình sáng bừng rực rỡ
                curStyle.Ambience = 0.65
                Try : curStyle.ShadowDensity = 0.55 : Catch : End Try

                Dim lNum As Integer = 1
                For Each lt As Light In curStyle.Lights
                    lt.Enabled = True
                    lt.Color = ThisApplication.TransientObjects.CreateColor(255, 255, 255)

                    If lNum = 1 Then
                        ' ĐÈN 1 (KEY LIGHT): Góc trước-trên-trái (Cường độ 220% - Rực sáng bề mặt chính)
                        lt.Intensity = 2.2
                        Try : lt.Direction = tg.CreateVector(-1.0, -1.2, -1.0) : Catch : End Try
                    ElseIf lNum = 2 Then
                        ' ĐÈN 2 (FILL LIGHT): Góc trước-trên-phải (Cường độ 180% - Bù sáng vùng khuất)
                        lt.Intensity = 1.8
                        Try : lt.Direction = tg.CreateVector(1.0, -1.0, -1.0) : Catch : End Try
                    ElseIf lNum = 3 Then
                        ' ĐÈN 3 (TOP LIGHT - SHADOW CASTER ABOVE): Chiếu thẳng đứng 90° từ đỉnh xuống (Cường độ 180% - Tạo bóng sàn Above chuẩn tụ đều dưới đáy)
                        lt.Intensity = 1.8

                    If rawT.Contains("COMP") OrElse rawT.Contains("ORGAN") OrElse rawT.Contains("WATERTREATMENT3D") Then
                        Try : tb.Color = oBlueCompany : Catch : End Try
                    End If
                Next
            End If
            tbDef.ExitEdit(True)
        Catch : End Try
    End If
    ' Force refresh so <COMP> iProperty tag renders in the title block
    oSheet.Update()
    oApp.UserInterfaceManager.DoEvents()

    ' =========================================================================
    ' 4. ADAPTIVE LAYOUT ALGORITHM  (Robust — works for ANY model size/shape)
    ' =========================================================================
    ' A3 ISO usable drawing area (cm):
    '   Left border  = 2.5, Right border = 1.0
    '   Top border   = 1.0, Bottom = title block ≈ 4.5 cm
    '   Title block occupies: X=24..41, Y=0..4.5
    '   Usable zone:  X: 2.5..41.0 | Y: 4.5..28.7
    '
    ' LAYOUT GRID:
    '   ┌──────────────┬──────────────┬────────────┐
    '   │  FRONT (Q1)  │  SIDE  (Q2)  │  3D  (Q3)  │  ← Upper row
    '   ├──────────────┴──────────────┤ Title Block │
    '   │  TOP PLAN (Q4)              │            │  ← Lower row
    '   └─────────────────────────────┴────────────┘
    ' =========================================================================
    Dim sheetW As Double = oSheet.Width    ' cm, A3 = 42.0
    Dim sheetH As Double = oSheet.Height   ' cm, A3 = 29.7
    Dim labelH As Double = 1.5             ' cm reserved below each view for label

    ' --- Zone boundaries (cm) ---
    Const areaX0 As Double = 2.5    ' left border
    Const areaX1 As Double = 41.0   ' right border
    Const areaY0 As Double = 4.5    ' above title block
    Dim   areaY1 As Double = sheetH - 1.0  ' below top border (~28.7)

    ' Vertical split: upper 62% for elevations, lower 38% for plan
    Dim totalAreaH As Double = areaY1 - areaY0          ' ~24.2 cm
    Dim upperZoneH As Double = totalAreaH * 0.62        ' ~15.0 cm
    Dim lowerZoneH As Double = totalAreaH * 0.38        ' ~9.2 cm
    Dim splitY     As Double = areaY0 + lowerZoneH      ' Y line splitting rows

    ' Column widths: 2D views left | 3D iso right
    Const col3D_X0 As Double = 29.5    ' 3D iso starts here
    Dim col2D_W    As Double = col3D_X0 - areaX0        ' ~27.0 cm for Front+Side
    Dim colFront_W As Double = col2D_W * 0.50 - 0.5    ' ~13.0 cm each
    Dim colSide_W  As Double = col2D_W * 0.50 - 0.5
    Dim col3D_W    As Double = areaX1 - col3D_X0        ' ~11.5 cm

    ' Zone heights minus label space
    Dim frontZoneH As Double = upperZoneH - labelH      ' ~13.5 cm usable for view
    Dim topZoneH   As Double = lowerZoneH - labelH      ' ~7.7 cm usable for view

    ' -------------------------------------------------------------------------
    ' 4a. GET MODEL DIMENSIONS (sorted small → large)
    ' -------------------------------------------------------------------------
    Dim dimX As Double = 30.0, dimY As Double = 100.0, dimZ As Double = 30.0
    Try
        Dim bBox As Box = Nothing
        If TypeOf oModelDoc Is PartDocument Then
            bBox = DirectCast(oModelDoc, PartDocument).ComponentDefinition.RangeBox
        ElseIf TypeOf oModelDoc Is AssemblyDocument Then
            bBox = DirectCast(oModelDoc, AssemblyDocument).ComponentDefinition.RangeBox
        End If
        If bBox IsNot Nothing Then
            dimX = Math.Abs(bBox.MaxPoint.X - bBox.MinPoint.X)
            dimY = Math.Abs(bBox.MaxPoint.Y - bBox.MinPoint.Y)
            dimZ = Math.Abs(bBox.MaxPoint.Z - bBox.MinPoint.Z)
        End If
    Catch : End Try

    ' Identify axes: Inventor Y=up → dimY is height in Front view
    ' Width of Front = max(dimX, dimZ), Height of Front = dimY
    Dim frontW_m As Double = Math.Max(dimX, dimZ)   ' model width in Front view
    Dim frontH_m As Double = dimY                    ' model height in Front view
    Dim sideW_m  As Double = Math.Min(dimX, dimZ)   ' model depth in Side view
    Dim sideH_m  As Double = dimY
    Dim topW_m   As Double = frontW_m               ' X extent in Top view
    Dim topH_m   As Double = sideW_m                ' Z extent in Top view

    ' Guard against zero or near-zero
    If frontW_m < 0.1 Then frontW_m = 1.0
    If frontH_m < 0.1 Then frontH_m = 1.0
    If sideW_m  < 0.1 Then sideW_m  = frontW_m
    If topH_m   < 0.1 Then topH_m   = 1.0

    ' -------------------------------------------------------------------------
    ' 4b. COMPUTE SCALE FROM 6 SIMULTANEOUS CONSTRAINTS
    ' -------------------------------------------------------------------------
    ' Each view must fit within its zone (both W and H), label included.
    Dim sc_FH As Double = frontZoneH / frontH_m   ' Front: height constraint
    Dim sc_FW As Double = colFront_W  / frontW_m  ' Front: width  constraint
    Dim sc_SH As Double = frontZoneH / sideH_m    ' Side:  height constraint
    Dim sc_SW As Double = colSide_W   / sideW_m   ' Side:  width  constraint
    Dim sc_TH As Double = topZoneH   / topH_m     ' Top:   height constraint
    Dim sc_TW As Double = colFront_W  / topW_m    ' Top:   width  constraint
    ' 3D iso diagonal ≈ sqrt(frontW^2 + frontH^2) × ~0.75 projected
    Dim isoDiag_m As Double = Math.Sqrt(frontW_m*frontW_m + frontH_m*frontH_m + sideW_m*sideW_m) * 0.6
    Dim sc_3D As Double = Math.Min(frontZoneH, col3D_W * 0.9) / Math.Max(isoDiag_m, 0.1)

    Dim rawScale As Double = Math.Min(sc_FH, Math.Min(sc_FW, Math.Min(sc_SH, Math.Min(sc_SW, Math.Min(sc_TH, Math.Min(sc_TW, sc_3D))))))
    rawScale = Math.Max(rawScale, 0.005) ' absolute minimum

    ' -------------------------------------------------------------------------
    ' 4c. SNAP TO ISO STANDARD SCALE (largest ISO scale ≤ rawScale)
    ' -------------------------------------------------------------------------
    Dim isoScales As Double() = {2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005}
    Dim S As Double = 0.005
    For Each sc As Double In isoScales
        If sc <= rawScale Then S = sc : Exit For
    Next

    ' -------------------------------------------------------------------------
    ' 4d. COMPUTE ACTUAL VIEW SIZES AT SCALE S
    '     and CENTER COORDINATES per zone
    ' -------------------------------------------------------------------------
    Dim vFW As Double = frontW_m * S  ' Front view rendered width (cm)
    Dim vFH As Double = frontH_m * S  ' Front view rendered height
    Dim vSW As Double = sideW_m  * S
    Dim vSH As Double = sideH_m  * S
    Dim vTW As Double = topW_m   * S
    Dim vTH As Double = topH_m   * S

    ' Zone centers
    Dim front_xCenter As Double = areaX0 + colFront_W / 2.0                ' center of front column
    Dim side_xCenter  As Double = areaX0 + colFront_W + 0.5 + colSide_W / 2.0  ' center of side column
    Dim iso_xCenter   As Double = col3D_X0 + col3D_W / 2.0                 ' center of 3D column

    Dim upper_yCenter As Double = splitY + upperZoneH / 2.0 - labelH / 2.0 ' center of upper row zone
    Dim lower_yCenter As Double = areaY0 + lowerZoneH / 2.0 - labelH / 2.0 ' center of lower row zone

    Dim x_front As Double = front_xCenter
    Dim x_side  As Double = side_xCenter
    Dim x_top   As Double = front_xCenter
    Dim x_iso   As Double = iso_xCenter
    Dim y_upper As Double = upper_yCenter
    Dim y_lower As Double = lower_yCenter

    ' -------------------------------------------------------------------------
    ' 5. DELETE OLD VIEWS & SETUP VIEW REPRESENTATION (Respect hidden items)
    ' -------------------------------------------------------------------------
    While oSheet.DrawingViews.Count > 0
        Try : oSheet.DrawingViews.Item(1).Delete() : Catch : Exit While : End Try
    End While

    ' Hide all work features, sketches, and construction geometry in drawing
    Try
        Dim oVis As ObjectVisibility = oDrawDoc.ObjectVisibility
        oVis.AllWorkFeatures = False
        oVis.Sketches = False
        oVis.Sketches3D = False
        oVis.Constructions = False
    Catch : End Try

    ' Turn off visibility of WorkSurfaces (ExtrusionSrf1, etc.) in model definition
    Try
        If TypeOf oModelDoc Is PartDocument Then
            Dim pDef As PartComponentDefinition = DirectCast(oModelDoc, PartDocument).ComponentDefinition
            For Each ws As WorkSurface In pDef.WorkSurfaces
                Try : ws.Visible = False : Catch : End Try
            Next
        ElseIf TypeOf oModelDoc Is AssemblyDocument Then
            Dim aDoc As AssemblyDocument = DirectCast(oModelDoc, AssemblyDocument)
            For Each occ As ComponentOccurrence In aDoc.ComponentDefinition.Occurrences
                If occ.DefinitionDocumentType = DocumentTypeEnum.kPartDocumentObject Then
                    Try
                        Dim pDef As PartComponentDefinition = DirectCast(occ.Definition, PartComponentDefinition)
                        For Each ws As WorkSurface In pDef.WorkSurfaces
                            Try : ws.Visible = False : Catch : End Try
                        Next
                    Catch : End Try
                End If
            Next
        End If
    Catch : End Try

    ' Resolve active Design View Representation from 3D model (Assembly/Part)
    Dim dvrName As String = ""
    Dim oViewOpts As NameValueMap = oApp.TransientObjects.CreateNameValueMap()
    oViewOpts.Add("IncludeSurfaceBodies", False)
    oViewOpts.Add("IncludeMeshBodies", False)
    Try
        If TypeOf oModelDoc Is AssemblyDocument Then
            Dim assemDoc As AssemblyDocument = DirectCast(oModelDoc, AssemblyDocument)
            If assemDoc.ComponentDefinition.RepresentationsManager.ActiveDesignViewRepresentation IsNot Nothing Then
                dvrName = assemDoc.ComponentDefinition.RepresentationsManager.ActiveDesignViewRepresentation.Name
                oViewOpts.Add("DesignViewRepresentation", dvrName)
                oViewOpts.Add("DesignViewAssociative", True)
            End If
        ElseIf TypeOf oModelDoc Is PartDocument Then
            Dim partDoc As PartDocument = DirectCast(oModelDoc, PartDocument)
            If partDoc.ComponentDefinition.RepresentationsManager.ActiveDesignViewRepresentation IsNot Nothing Then
                dvrName = partDoc.ComponentDefinition.RepresentationsManager.ActiveDesignViewRepresentation.Name
                oViewOpts.Add("DesignViewRepresentation", dvrName)
                oViewOpts.Add("DesignViewAssociative", True)
            End If
        End If
    Catch : End Try

    ' Helper to create BaseView with DVR and hidden items filtering
    Dim CreateView As Func(Of Point2d, Double, ViewOrientationTypeEnum, DrawingViewStyleEnum, DrawingView) =
        Function(pt As Point2d, sc As Double, ori As ViewOrientationTypeEnum, sty As DrawingViewStyleEnum) As DrawingView
            Dim dv As DrawingView = Nothing
            Try
                If dvrName <> "" Then
                    dv = oSheet.DrawingViews.AddBaseView(oModelDoc, pt, sc, ori, sty, dvrName, Nothing, oViewOpts)
                Else
                    dv = oSheet.DrawingViews.AddBaseView(oModelDoc, pt, sc, ori, sty)
                End If
            Catch
                Try : dv = oSheet.DrawingViews.AddBaseView(oModelDoc, pt, sc, ori, sty) : Catch : End Try
            End Try

            If dv IsNot Nothing AndAlso dvrName <> "" Then
                Try : dv.SetDesignViewRepresentation(dvrName, True) : Catch : End Try
            End If
            Return dv
        End Function

    ' Declare view variables upfront (needed by closure)
    Dim oFront As DrawingView = Nothing
    Dim oSide  As DrawingView = Nothing
    Dim oTop   As DrawingView = Nothing
    Dim oIso   As DrawingView = Nothing

    ' Helper to hide all WorkSurfaces (ExtrusionSrf1, etc.) & hidden components across all views
    Dim HideSurfacesAndHiddenComponents As Action =
        Sub()
            ' 1. Hide Assembly Occurrences if hidden/suppressed
            If TypeOf oModelDoc Is AssemblyDocument Then
                Try
                    Dim assemDoc As AssemblyDocument = DirectCast(oModelDoc, AssemblyDocument)
                    For Each occ As ComponentOccurrence In assemDoc.ComponentDefinition.Occurrences
                        If Not occ.Visible OrElse occ.Suppressed Then
                            Try
                                If oFront IsNot Nothing Then oFront.SetVisibility(occ, False)
                            Catch : End Try
                            Try
                                If oSide IsNot Nothing Then oSide.SetVisibility(occ, False)
                            Catch : End Try
                            Try
                                If oTop IsNot Nothing Then oTop.SetVisibility(occ, False)
                            Catch : End Try
                            Try
                                If oIso IsNot Nothing Then oIso.SetVisibility(occ, False)
                            Catch : End Try
                        End If
                    Next
                Catch : End Try
            End If

            ' 2. Hide Drawing Curves of all WorkSurfaces (ExtrusionSrf1, etc.) & non-solid SurfaceBodies
            Dim vList As New List(Of DrawingView)()
            If oFront IsNot Nothing Then vList.Add(oFront)
            If oSide  IsNot Nothing Then vList.Add(oSide)
            If oTop   IsNot Nothing Then vList.Add(oTop)
            If oIso   IsNot Nothing Then vList.Add(oIso)

            For Each dv As DrawingView In vList
                Try
                    Dim refDoc As Document = dv.ReferencedDocumentDescriptor.ReferencedDocument
                    If TypeOf refDoc Is PartDocument Then
                        Dim pDef As PartComponentDefinition = DirectCast(refDoc, PartDocument).ComponentDefinition
                        For Each ws As WorkSurface In pDef.WorkSurfaces
                            Try
                                Dim curves As DrawingCurvesEnumerator = dv.DrawingCurves(ws)
                                If curves IsNot Nothing Then
                                    For Each dc As DrawingCurve In curves
                                        For Each seg As DrawingCurveSegment In dc.Segments
                                            seg.Visible = False
                                        Next
                                    Next
                                End If
                            Catch : End Try
                        Next
                        For Each sb As SurfaceBody In pDef.SurfaceBodies
                            If Not sb.IsSolid Then
                                Try
                                    Dim curves As DrawingCurvesEnumerator = dv.DrawingCurves(sb)
                                    If curves IsNot Nothing Then
                                        For Each dc As DrawingCurve In curves
                                            For Each seg As DrawingCurveSegment In dc.Segments
                                                seg.Visible = False
                                            Next
                                        Next
                                    End If
                                Catch : End Try
                            End If
                        Next
                    ElseIf TypeOf refDoc Is AssemblyDocument Then
                        Dim aDef As AssemblyComponentDefinition = DirectCast(refDoc, AssemblyDocument).ComponentDefinition
                        For Each occ As ComponentOccurrence In aDef.Occurrences
                            If occ.DefinitionDocumentType = DocumentTypeEnum.kPartDocumentObject Then
                                Try
                                    Dim occPDef As PartComponentDefinition = DirectCast(occ.Definition, PartComponentDefinition)
                                    For Each ws As WorkSurface In occPDef.WorkSurfaces
                                        Try
                                            Dim curves As DrawingCurvesEnumerator = dv.DrawingCurves(ws)
                                            If curves IsNot Nothing Then
                                                For Each dc As DrawingCurve In curves
                                                    For Each seg As DrawingCurveSegment In dc.Segments
                                                        seg.Visible = False
                                                    Next
                                                Next
                                            End If
                                        Catch : End Try
                                    Next
                                    For Each sb As SurfaceBody In occPDef.SurfaceBodies
                                        If Not sb.IsSolid Then
                                            Try
                                                Dim curves As DrawingCurvesEnumerator = dv.DrawingCurves(sb)
                                                If curves IsNot Nothing Then
                                                    For Each dc As DrawingCurve In curves
                                                        For Each seg As DrawingCurveSegment In dc.Segments
                                                            seg.Visible = False
                                                        Next
                                                    Next
                                                End If
                                            Catch : End Try
                                        End If
                                    Next
                                Catch : End Try
                            End If
                        Next
                    End If
                Catch : End Try
            Next
        End Sub

    ' -------------------------------------------------------------------------
    ' 6. CREATE 4 INDEPENDENT BASE VIEWS (Hidden Line Removed for 2D Views)
    ' -------------------------------------------------------------------------
    oFront = CreateView(tg.CreatePoint2d(x_front, y_upper), S, ViewOrientationTypeEnum.kFrontViewOrientation,     DrawingViewStyleEnum.kHiddenLineRemovedDrawingViewStyle)
    oSide  = CreateView(tg.CreatePoint2d(x_side,  y_upper), S, ViewOrientationTypeEnum.kRightViewOrientation,      DrawingViewStyleEnum.kHiddenLineRemovedDrawingViewStyle)
    oTop   = CreateView(tg.CreatePoint2d(x_top,   y_lower), S, ViewOrientationTypeEnum.kTopViewOrientation,         DrawingViewStyleEnum.kHiddenLineRemovedDrawingViewStyle)
    oIso   = CreateView(tg.CreatePoint2d(x_iso,   y_upper), S, ViewOrientationTypeEnum.kIsoTopRightViewOrientation, DrawingViewStyleEnum.kShadedDrawingViewStyle)

    HideSurfacesAndHiddenComponents()

    oSheet.Update()
    oDrawDoc.Update()
    oApp.UserInterfaceManager.DoEvents()

    ' =========================================================================
    ' 7. COLLISION CHECK + AUTO-RETRY (reduce ISO scale one step if overlap)
    ' =========================================================================
    Dim maxRetry As Integer = 4
    Dim retryIdx As Integer = 0
    For idxSc As Integer = 0 To isoScales.Length - 1
        If isoScales(idxSc) = S Then retryIdx = idxSc : Exit For
    Next

    For attempt As Integer = 1 To maxRetry
        Dim hasOverlap As Boolean = False
        Dim outOfBounds As Boolean = False

        ' Read ACTUAL rendered sizes from Inventor after update
        Dim fW As Double = If(oFront IsNot Nothing, oFront.Width,  0)
        Dim fH As Double = If(oFront IsNot Nothing, oFront.Height, 0)
        Dim sW As Double = If(oSide  IsNot Nothing, oSide.Width,   0)
        Dim sH As Double = If(oSide  IsNot Nothing, oSide.Height,  0)
        Dim tW As Double = If(oTop   IsNot Nothing, oTop.Width,    0)
        Dim tH As Double = If(oTop   IsNot Nothing, oTop.Height,   0)
        Dim iW As Double = If(oIso   IsNot Nothing, oIso.Width,    0)
        Dim iH As Double = If(oIso   IsNot Nothing, oIso.Height,   0)

        ' Gap check: Front right edge vs Side left edge
        Dim frontRightEdge As Double = If(oFront IsNot Nothing, oFront.Center.X + fW/2.0, 0)
        Dim sideLeftEdge   As Double = If(oSide  IsNot Nothing, oSide.Center.X  - sW/2.0, 99)
        If frontRightEdge >= sideLeftEdge - 0.3 Then hasOverlap = True  ' < 3mm gap

        ' Gap check: Side right edge vs 3D left edge
        Dim sideRightEdge As Double = If(oSide IsNot Nothing, oSide.Center.X + sW/2.0, 0)
        Dim isoLeftEdge   As Double = If(oIso  IsNot Nothing, oIso.Center.X  - iW/2.0, 99)
        If sideRightEdge >= isoLeftEdge - 0.3 Then hasOverlap = True

        ' Check Top view vs bottom border
        Dim topBottomEdge As Double = If(oTop IsNot Nothing, oTop.Center.Y - tH/2.0, 99)
        If topBottomEdge <= areaY0 Then outOfBounds = True

        ' Check any view exceeds top border
        Dim upperTopEdge As Double = 0.0
        For Each view As DrawingView In oSheet.DrawingViews
            If view.Center.Y + view.Height/2.0 > upperTopEdge Then upperTopEdge = view.Center.Y + view.Height/2.0
        Next
        If upperTopEdge >= areaY1 Then outOfBounds = True

        If Not hasOverlap AndAlso Not outOfBounds Then Exit For  ' All good!

        ' --- Reduce scale one ISO step and redo ---
        retryIdx = retryIdx + 1
        If retryIdx >= isoScales.Length - 1 Then Exit For
        S = isoScales(retryIdx)

        ' Recompute positions with new S (views are SMALLER → re-center in same zones)
        While oSheet.DrawingViews.Count > 0
            Try : oSheet.DrawingViews.Item(1).Delete() : Catch : Exit While : End Try
        End While
        oFront = Nothing : oSide = Nothing : oTop = Nothing : oIso = Nothing

        oFront = CreateView(tg.CreatePoint2d(x_front, y_upper), S, ViewOrientationTypeEnum.kFrontViewOrientation,     DrawingViewStyleEnum.kHiddenLineRemovedDrawingViewStyle)
        oSide  = CreateView(tg.CreatePoint2d(x_side,  y_upper), S, ViewOrientationTypeEnum.kRightViewOrientation,      DrawingViewStyleEnum.kHiddenLineRemovedDrawingViewStyle)
        oTop   = CreateView(tg.CreatePoint2d(x_top,   y_lower), S, ViewOrientationTypeEnum.kTopViewOrientation,         DrawingViewStyleEnum.kHiddenLineRemovedDrawingViewStyle)
        oIso   = CreateView(tg.CreatePoint2d(x_iso,   y_upper), S, ViewOrientationTypeEnum.kIsoTopRightViewOrientation, DrawingViewStyleEnum.kShadedDrawingViewStyle)

        HideSurfacesAndHiddenComponents()

        oSheet.Update() : oDrawDoc.Update() : oApp.UserInterfaceManager.DoEvents()
    Next

    ' -------------------------------------------------------------------------
    ' 8. FORCE LOCK POSITIONS (second pass — fix any drift after update)
    ' -------------------------------------------------------------------------
    Try
        If oFront IsNot Nothing Then oFront.Position = tg.CreatePoint2d(x_front, y_upper)
    Catch : End Try
    Try
        If oSide IsNot Nothing Then oSide.Position = tg.CreatePoint2d(x_side, y_upper)
    Catch : End Try
    Try
        If oTop IsNot Nothing Then oTop.Position = tg.CreatePoint2d(x_top, y_lower)
    Catch : End Try
    Try
        If oIso IsNot Nothing Then oIso.Position = tg.CreatePoint2d(x_iso, y_upper)
    Catch : End Try

    Dim scaleStr As String = If(S >= 1.0, CInt(S).ToString() & ":1", "1:" & CInt(Math.Round(1.0 / S)).ToString())
    exportResults.Add("ISO A3 4-Quadrant (Scale " & scaleStr & ", Collision-safe adaptive layout)")
    oApp.StatusBarText = "[WT3D V9.4]: Layout OK — Scale " & scaleStr

    ' -------------------------------------------------------------------------
    ' 9. SET LABELS — BOLD BLUE (#0050C8) via Drawing Standards & Styles
    ' -------------------------------------------------------------------------
    Dim labelGap As Double = 1.2
    Dim oBlue As Color = oApp.TransientObjects.CreateColor(0, 80, 200)

    ' 1. Set Color on ALL Label TextStyles in drawing (Label Text (ISO), Label Text (ANSI), etc.)
    Try
        For Each ts As TextStyle In oDrawDoc.StylesManager.TextStyles
            If ts.Name.ToLower().Contains("label") Then
                Try : ts.Color = oBlue : Catch : End Try
            End If
        Next
    Catch : End Try

    ' 2. Target specific ISO TextStyles directly
    Try : oDrawDoc.StylesManager.TextStyles.Item("Label Text (ISO)").Color = oBlue : Catch : End Try
    Try : oDrawDoc.StylesManager.TextStyles.Item("Label Text (ANSI)").Color = oBlue : Catch : End Try

    ' 3. Set Color on Active Drawing Standard's ViewLabel TextStyle & Layer (for "By Layer" inheritance)
    Try
        Dim actStd As DrawingStandardStyle = oDrawDoc.StylesManager.ActiveStandardStyle
        If actStd IsNot Nothing AndAlso actStd.ActiveObjectDefaults IsNot Nothing Then
            Try : actStd.ActiveObjectDefaults.ViewLabelTextStyle.Color = oBlue : Catch : End Try
            Try : actStd.ActiveObjectDefaults.ViewLabelLayer.Color = oBlue : Catch : End Try
        End If
    Catch : End Try

    ' 4. Assign FormattedText and Position to each View Label
    Dim SetLabel As Action(Of DrawingView, String) =
        Sub(dv As DrawingView, txt As String)
            If dv Is Nothing Then Return
            Try
                dv.ShowLabel = True
                dv.Label.FormattedText = "<StyleOverride Bold='True'>" & txt & "</StyleOverride><Br/>SCALE <DrawingViewScale/>"
                dv.Label.Position = tg.CreatePoint2d(dv.Center.X, dv.Center.Y - dv.Height / 2.0 - labelGap)
            Catch : End Try
        End Sub

    SetLabel(oFront, "FRONT ELEVATION")
    SetLabel(oSide,  "RIGHT SIDE ELEVATION")
    SetLabel(oTop,   "TOP PLAN VIEW")
    SetLabel(oIso,   "ISOMETRIC 3D VIEW (3/4)")

    oSheet.Update()
    oDrawDoc.Update()
    oApp.UserInterfaceManager.DoEvents()

    ' Copyright footer (Blue)
    Try
        Dim gn As GeneralNote = oSheet.DrawingNotes.GeneralNotes.AddFitted(tg.CreatePoint2d(2.5, 0.8), "WaterTreatment3D  |  Designer: PHAN TRONG TAN (@tanphan1105)  |  ISO Standard CAD")
        gn.Color = oBlue
    Catch : End Try

    exportResults.Add("ISO A3 4-Quadrant Views (All 4 AddBaseView, Scale " & scaleStr & ")")

    oSheet.Update()
    oDrawDoc.Update()
    oApp.UserInterfaceManager.DoEvents()

    ' -------------------------------------------------------------------------
    ' 10. 4K BITMAP CAPTURE
    ' -------------------------------------------------------------------------
    Dim oDrawView As Inventor.View = oApp.ActiveView
    oDrawView.Fit()
    oDrawView.Update()
    oApp.UserInterfaceManager.DoEvents()

    Dim img4K As String = System.IO.Path.Combine(imgDir, modelBaseName & "_Drawing_Sheet_4K.png")
    Try
        oDrawView.SaveAsBitmap(img4K, 3840, 2160)
        If System.IO.File.Exists(img4K) Then
            System.IO.File.Copy(img4K, System.IO.Path.Combine(exportDir, modelBaseName & "_Drawing_Sheet.png"), True)
            exportResults.Add("Drawing Sheet 4K PNG")
        End If
    Catch : End Try

    ' -------------------------------------------------------------------------
    ' 11. EXPORT DWG + PDF
    ' -------------------------------------------------------------------------
    Dim dwgPath As String = System.IO.Path.Combine(exportDir, modelBaseName & "_Drawing_Sheet.dwg")
    Try
        oDrawDoc.SaveAs(dwgPath, True)
        If System.IO.File.Exists(dwgPath) Then exportResults.Add("AutoCAD DWG (.dwg)")
    Catch : End Try

    Dim pdfPath As String = System.IO.Path.Combine(exportDir, modelBaseName & "_Drawing_Sheet.pdf")
    Try
        For Each addIn As ApplicationAddIn In oApp.ApplicationAddIns
            If addIn.ClassIdString = "{0AC6FD96-2F4D-42CE-8BE0-8AEA580399E4}" Then
                Dim pdfAddIn As TranslatorAddIn = DirectCast(addIn, TranslatorAddIn)
                Dim ctx As TranslationContext = oApp.TransientObjects.CreateTranslationContext()
                ctx.Type = IOMechanismEnum.kFileBrowseIOMechanism
                Dim opts As NameValueMap = oApp.TransientObjects.CreateNameValueMap()
                Dim dm As DataMedium = oApp.TransientObjects.CreateDataMedium()
                dm.FileName = pdfPath
                If pdfAddIn.HasSaveCopyAsOptions(oDrawDoc, ctx, opts) Then
                    opts.Value("Vector_Resolution") = 400
                    opts.Value("Sheet_Range") = PrintRangeEnum.kPrintAllSheets
                    pdfAddIn.SaveCopyAs(oDrawDoc, ctx, opts, dm)
                    If System.IO.File.Exists(pdfPath) Then exportResults.Add("Vector PDF (.pdf)")
                End If
                Exit For
            End If
        Next
    Catch : End Try

    ' -------------------------------------------------------------------------
    ' 12. EXPORT STEP
    ' -------------------------------------------------------------------------
    Try
