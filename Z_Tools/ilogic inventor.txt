public class CoreFamilyEngine
{
    private readonly UIApplication _uiApp;

    public CoreFamilyEngine(UIApplication uiApp)
    {
        _uiApp = uiApp;
    }

    public void Process(string satPath, SatMetadata meta, string templatePath)
    {
        Document famDoc = _uiApp.Application.NewFamilyDocument(templatePath);

        TransactionGroup tg = new TransactionGroup(famDoc, "SAT_TO_FAMILY");
        tg.Start();

        try
        {
            FreeFormElement freeForm = null;

            using (Transaction tx = new Transaction(famDoc, "Create Geometry"))
            {
                tx.Start();

                freeForm = ImportSATToFreeForm(famDoc, satPath);

                tx.Commit();
            }

            using (Transaction tx2 = new Transaction(famDoc, "Attach Connectors + Params"))
            {
                tx2.Start();

                AttachConnectors(famDoc, freeForm, meta);
                BindParameters(famDoc, meta);

                tx2.Commit();
            }

            tg.Assimilate();
        }
        catch
        {
            tg.RollBack();
            throw;
        }

        ExportFamily(famDoc, satPath);
    }
}
🧱 3. SAT → FREEFORM ELEMENT (CORE LOGIC)
private FreeFormElement ImportSATToFreeForm(Document doc, string satPath)
{
    // STEP 1: Import SAT as DirectShape container
    ImportOptions opt = new ImportOptions
    {
        Placement = ImportPlacement.Origin
    };

    ElementId importId = doc.Import(satPath, opt);

    Element imported = doc.GetElement(importId);

    GeometryElement geo = imported.get_Geometry(new Options());

    Solid solid = ExtractFirstSolid(geo);

    if (solid == null)
        throw new Exception("No valid solid found in SAT");

    // STEP 2: Create FreeFormElement from solid
    FreeFormElement freeForm = FreeFormElement.Create(doc, solid);

    // STEP 3: Delete imported placeholder
    doc.Delete(importId);

    return freeForm;
}
🔍 4. SOLID EXTRACTION (SAFE VERSION)
private Solid ExtractFirstSolid(GeometryElement geo)
{
    foreach (GeometryObject obj in geo)
    {
        if (obj is Solid s && s.Volume > 0)
            return s;

        if (obj is GeometryInstance gi)
        {
            GeometryElement instGeo = gi.GetInstanceGeometry();
            Solid inner = ExtractFirstSolid(instGeo);
            if (inner != null)
                return inner;
        }
    }

    return null;
}

private void AttachConnectors(Document doc, FreeFormElement ff, SatMetadata meta)
{
    var geom = ff.get_Geometry(new Options());

    Solid solid = ExtractFirstSolid(geom);

    Face inletFace = FindBestFace(solid, meta.Inlet.Direction);
    Face outletFace = FindBestFace(solid, meta.Outlet.Direction);

    using (Transaction tx = new Transaction(doc, "Connectors"))
    {
        tx.Start();

        CreateConnector(doc, inletFace, meta.DnIn, true);
        CreateConnector(doc, outletFace, meta.DnOut, false);

        tx.Commit();
    }
}
🎯 6. FACE SELECTION LOGIC (DIRECTION MATCHING)
private Face FindBestFace(Solid solid, double[] dir)
{
    XYZ targetDir = new XYZ(dir[0], dir[1], dir[2]).Normalize();

    Face best = null;
    double maxDot = -1;

    foreach (Face f in solid.Faces)
    {
        XYZ normal = GetFaceNormal(f);

        double dot = Math.Abs(normal.DotProduct(targetDir));

        if (dot > maxDot)
        {
            maxDot = dot;
            best = f;
        }
    }

    return best;
}

private void SetParam(FamilyManager fm, string name, double value)
{
    FamilyParameter p = fm.get_Parameter(name);

    if (p != null)
        fm.Set(p, value);
}
